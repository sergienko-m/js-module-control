Завдання 1 — summarizeNumbers()

Приймає масив чисел, повертає статистику. Цикл for без HOF.

function summarizeNumbers(numbers) {
  if (!numbers.length) return { count:0, sum:0, evenCount:0, max:undefined, category:'empty' };

  let sum=0, evenCount=0, max=numbers[0];

  for (let i=0; i < numbers.length; i++) {
    sum += numbers[i];
    if (numbers[i] % 2 === 0) evenCount++;
    if (numbers[i] > max) max = numbers[i];
  }

  return { count: numbers.length, sum, evenCount, max,
    category: sum > 0 ? 'positive' : 'non-positive' };
}

Завдання 2 — processProducts()

Масив товарів → filter, reduce, map.

function processProducts(products) {
  const inStock   = products.filter(p => p.inStock);
  const available = inStock.map(p => p.name);
  const totalPrice = inStock.reduce((s,p) => s+p.price, 0);
  const cheapest  = inStock.reduce((m,p) => p.price < m.price ? p : m)?.name;
  const priceList = products.map(p => `${p.name} — ${p.price} грн`);
  return { available, totalPrice, cheapest, priceList };
}

Завдання 3 — createApiClient()

Замикання + async/await + fetch. Реальні запити до jsonplaceholder.typicode.com

function createApiClient(baseUrl) {
  let requestCount = 0; // приватний лічильник у замиканні

  return {
    async get(path) {
      requestCount++;
      try {
        const response = await fetch(baseUrl + path);
        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        return await response.json();
      } catch {
        return { error: 'Запит не вдався' };
      }
    },
    getRequestCount() { return requestCount; }
  };
}

Завдання 4 — Task + TodoList + DOM

Класи, ООП, делегування подій.

class Task {
  constructor(id, text) { this.id=id; this.text=text; this.done=false; }
  toggle() { this.done = !this.done; }
}
class TodoList {
  #tasks=[]; #nextId=1;
  add(text)    { const t=new Task(this.#nextId++, text); this.#tasks.push(t); return t; }
  remove(id)   { this.#tasks = this.#tasks.filter(t => t.id !== id); }
  getActive()  { return this.#tasks.filter(t => !t.done); }
  getAll()     { return [...this.#tasks]; }
  findById(id) { return this.#tasks.find(t => t.id===id); }
}
